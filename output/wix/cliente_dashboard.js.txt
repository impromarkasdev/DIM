// Código completo del Cliente Dashboard
import { processSearchAndRedact, getDocumentJobStatus, combineRedactionJob } from 'backend/api';
import { local, session } from 'wix-storage';
import wixLocation from 'wix-location';

const JOB_CACHE_KEY = 'dim_last_job';
const JOB_CACHE_MAX_AGE_MS = 6 * 60 * 60 * 1000;
const JOB_POLL_TIMEOUT_MS = 8 * 60 * 1000;
const MAX_REFERENCES = 30;

let searchInProgress = false;

$w.onReady(function () {
    const token = session.getItem('dim_jwt') || local.getItem('jwt_token') || local.getItem('userToken');
    const role = String(session.getItem('dim_role') || local.getItem('user_role') || '').trim().toLowerCase();
    if (!token || role !== 'client') {
        wixLocation.to('/declaraciones');
        return;
    }

    $w('#btnDownloadClient').hide();
    $w('#txtClientStatus').text = '';
    $w('#btnClientSearch').onClick(async () => {
        if (searchInProgress) return;
        const references = normalizeReferences($w('#inputClientRef').value);
        hideDownloads();

        if (!references.length) {
            setClientStatus('Ingresa al menos una referencia. Puedes separarlas por líneas, comas o punto y coma.');
            return;
        }
        if (references.length > MAX_REFERENCES) {
            setClientStatus(`Puedes consultar hasta ${MAX_REFERENCES} referencias por solicitud.`);
            return;
        }

        searchInProgress = true;
        $w('#btnClientSearch').disable();
        setClientStatus(`Progreso: 0 de ${references.length} referencias. Preparando la consulta…`);

        try {
            const job = await submitOrResumeRedaction(token, references);
            if (job.pending) {
                setClientStatus(`La solicitud continúa en segundo plano (seguimiento ${job.jobId}). No la envíes de nuevo; vuelve a consultar en unos momentos.`);
                return;
            }

            const results = Array.isArray(job.resultados) ? job.resultados : [];
            const successful = results.filter(item => item?.estado === 'Exito');
            const ready = successful.filter(item => item.downloadUrl);
            if (successful.length) {
                let downloadUrl = ready[0]?.downloadUrl || '';
                let combinedSucceeded = false;
                let uniqueDeclarationCount = successful.length;
                if (successful.length > 1 || !downloadUrl) {
                    setClientStatus(`Se encontraron ${successful.length} resultados. Unificando declaraciones repetidas…`);
                    try {
                        const combined = await combineRedactionJob(token, job.jobId);
                        downloadUrl = combined?.downloadUrl || '';
                        uniqueDeclarationCount = Number(combined?.combinedCount || successful.length);
                        combinedSucceeded = Boolean(downloadUrl);
                    } catch (error) {
                        console.error('No se pudo unificar el PDF:', error);
                    }
                }
                if (!downloadUrl) {
                    saveJobCache(makeQueryKey(references), job.jobId);
                    setClientStatus('El documento quedó procesado, pero el enlace no está disponible todavía. Vuelve a consultar en unos momentos.');
                    return;
                }
                $w('#btnDownloadClient').link = downloadUrl;
                $w('#btnDownloadClient').target = '_blank';
                $w('#btnDownloadClient').show();

                saveJobCache(makeQueryKey(references), job.jobId);
                const notFound = results.filter(item => item?.estado === 'Pendiente (No encontrado)').length;
                const failed = results.filter(item => item?.estado === 'Error' || item?.estado === 'failed').length;
                const exceptions = [
                    notFound ? `No se encontró documento para ${notFound} referencia(s).` : '',
                    failed ? `${failed} referencia(s) requieren revisión por un error de procesamiento.` : ''
                ].filter(Boolean).join(' ');
                setClientStatus(successful.length === 1
                    ? `Consulta completada: 1 de ${references.length} documento(s) listo(s). ${exceptions}`.trim()
                    : combinedSucceeded
                        ? `Consulta completada: ${uniqueDeclarationCount} declaración(es) única(s) unificada(s) en un PDF para ${successful.length} referencia(s). ${exceptions}`.trim()
                        : `Se procesaron ${successful.length} documentos. El botón descarga el primero; no se pudo generar el combinado. ${exceptions}`.trim());
                return;
            }

            clearJobCache();
            const notFoundCount = results.filter(item => item?.estado === 'Pendiente (No encontrado)').length;
            const failedItems = results.filter(item => item?.estado === 'Error' || item?.estado === 'failed');
            const detail = failedItems.find(item => item?.detalle)?.detalle || job.lastError;
            setClientStatus(failedItems.length
                ? `${failedItems.length} referencia(s) no pudieron procesarse. ${detail || 'El administrador puede revisar el detalle del trabajo.'}`
                : notFoundCount
                    ? `El proceso terminó. No se encontró una declaración para ${notFoundCount} referencia(s).`
                    : 'El proceso terminó sin generar documentos descargables. Inténtalo nuevamente o contacta al administrador.');
        } catch (error) {
            console.error('Error al consultar referencias:', error);
            setClientStatus(error.message || 'No fue posible completar la consulta. Inténtalo nuevamente.');
        } finally {
            searchInProgress = false;
            $w('#btnClientSearch').enable();
        }
    });

    $w('#btnLogoutClient').onClick(() => {
        session.removeItem('dim_jwt');
        session.removeItem('dim_role');
        local.removeItem('jwt_token');
        local.removeItem('userToken');
        local.removeItem('user_role');
        session.removeItem(JOB_CACHE_KEY);
        wixLocation.to('/declaraciones');
    });
});

async function submitOrResumeRedaction(token, references) {
    const queryKey = makeQueryKey(references);
    const cached = readJobCache(queryKey);
    if (cached?.jobId) {
        const existing = await getDocumentJobStatus(token, cached.jobId);
        if (existing.success) {
            if (isTerminal(existing.status)) return existing;
            return waitForJob(token, cached.jobId, references.length);
        }
        if (existing.httpStatus !== 404) {
            throw new Error(existing.message || 'No fue posible consultar el trabajo anterior.');
        }
    }

    const idempotencyKey = cached?.idempotencyKey || makeIdempotencyKey('redact');
    saveJobCache(queryKey, cached?.jobId || '', idempotencyKey);
    const accepted = await processSearchAndRedact(token, references, idempotencyKey);
    if (!accepted.success || !accepted.jobId) {
        throw new Error(accepted.message || 'No fue posible iniciar la consulta.');
    }
    saveJobCache(queryKey, accepted.jobId, idempotencyKey);
    return waitForJob(token, accepted.jobId, references.length);
}

async function waitForJob(token, jobId, requestedTotal) {
    const deadline = Date.now() + JOB_POLL_TIMEOUT_MS;
    let delayMs = 2000;
    while (Date.now() < deadline) {
        const job = await getDocumentJobStatus(token, jobId);
        if (!job.success) {
            if ([429, 502, 503, 504].includes(job.httpStatus)) {
                await delay(delayMs);
                delayMs = Math.min(5000, delayMs + 500);
                continue;
            }
            throw new Error(job.message || 'No fue posible consultar el trabajo.');
        }
        const total = Number(job.total || requestedTotal || 0);
        const completed = Number(job.completed || 0);
        setClientStatus(job.queueDispatchPending
            ? `Progreso: ${Math.min(completed, total)} de ${total}. La cola aún no confirma los pendientes; se reintentará automáticamente…`
            : `Progreso: ${Math.min(completed, total)} de ${total} referencias procesadas…`);
        if (isTerminal(job.status)) return job;
        await delay(delayMs);
        delayMs = Math.min(5000, delayMs + 500);
    }
    return { success: false, pending: true, jobId };
}

function normalizeReferences(value) {
    const seen = new Set();
    return String(value || '')
        .split(/[\n,;]+/)
        .map(item => item.trim())
        .filter(item => {
            const key = item.toUpperCase();
            if (!key || seen.has(key)) return false;
            seen.add(key);
            return true;
        });
}

function hideDownloads() {
    $w('#btnDownloadClient').hide();
}

function isTerminal(status) {
    return ['completed', 'partial', 'failed'].includes(String(status || '').toLowerCase());
}

function makeQueryKey(references) {
    return JSON.stringify(references.map(value => String(value).trim().toUpperCase()).sort());
}

function makeIdempotencyKey(prefix) {
    return `${prefix}-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`;
}

function readJobCache(queryKey) {
    try {
        const raw = session.getItem(JOB_CACHE_KEY);
        if (!raw) return null;
        const cache = JSON.parse(raw);
        if (cache.queryKey !== queryKey || Date.now() - Number(cache.savedAt || 0) > JOB_CACHE_MAX_AGE_MS) {
            return null;
        }
        return cache;
    } catch (_) {
        return null;
    }
}

function saveJobCache(queryKey, jobId, idempotencyKey) {
    session.setItem(JOB_CACHE_KEY, JSON.stringify({
        queryKey,
        jobId,
        idempotencyKey,
        savedAt: Date.now()
    }));
}

function clearJobCache() {
    session.removeItem(JOB_CACHE_KEY);
}

function delay(milliseconds) {
    return new Promise(resolve => setTimeout(resolve, milliseconds));
}

function setClientStatus(message) {
    $w('#txtClientStatus').text = message;
}
